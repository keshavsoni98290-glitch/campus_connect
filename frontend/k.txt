/* =========================================================
   PAGE NAVIGATION
   ========================================================= */

let currentPage = "home";

function showPage(pageName) {

    // Hide all pages
    document.querySelectorAll(".page").forEach(page => {
        page.classList.remove("active-page");
    });

    // Show selected page
    const selectedPage = document.getElementById(pageName + "-page");

    if (selectedPage) {
        selectedPage.classList.add("active-page");
        currentPage = pageName;
    }

    // Back button
    const backButton = document.getElementById("backButton");

    if (backButton) {
        if (pageName === "home") {
            backButton.style.visibility = "hidden";
        } else {
            backButton.style.visibility = "visible";
        }
    }

    // Close team panels when changing page
    if (pageName !== "discover") {
        closeTeamPanels();
    }
}


/* =========================================================
   BACK BUTTON
   ========================================================= */

function goBack() {

    if (currentPage === "home") {
        return;
    }

    showPage("home");
}


/* =========================================================
   NAVBAR BUTTONS
   ========================================================= */

document.addEventListener("DOMContentLoaded", function () {

    // Navbar navigation
    document.querySelectorAll(".nav-links button").forEach(button => {

        button.addEventListener("click", function () {

            const page = this.getAttribute("data-page");

            if (page) {
                showPage(page);
            }

        });

    });

    // Start with home page
    showPage("home");


    /* =====================================================
       STUDENT SEARCH
       ===================================================== */

    const studentSearch = document.getElementById("studentSearch");

    if (studentSearch) {

        studentSearch.addEventListener("input", function () {

            filterStudents();

        });

    }


    /* =====================================================
       STUDENT FILTER
       ===================================================== */

    const studentFilter = document.getElementById("studentFilter");

    if (studentFilter) {

        studentFilter.addEventListener("change", function () {

            filterStudents();

        });

    }


    /* =====================================================
       COMMUNITY SEARCH
       ===================================================== */

    const communitySearch = document.getElementById("communitySearch");

    if (communitySearch) {

        communitySearch.addEventListener("input", function () {

            renderCommunities();

        });

    }


    /* =====================================================
       COMMUNITY CATEGORY BUTTONS
       ===================================================== */

    document.querySelectorAll(".category").forEach(button => {

        button.addEventListener("click", function () {

            document.querySelectorAll(".category").forEach(btn => {
                btn.classList.remove("active");
            });

            this.classList.add("active");

            renderCommunities();

        });

    });


    /* =====================================================
       CREATE COMMUNITY BUTTONS
       ===================================================== */

    const createCommunityBtn =
        document.getElementById("createCommunityBtn");

    const createCommunityBottom =
        document.getElementById("createCommunityBottom");

    if (createCommunityBtn) {
        createCommunityBtn.addEventListener(
            "click",
            createCommunity
        );
    }

    if (createCommunityBottom) {
        createCommunityBottom.addEventListener(
            "click",
            createCommunity
        );
    }


    /* =====================================================
       INITIAL DATA
       ===================================================== */

    loadCommunities();

});


/* =========================================================
   TEAM PANEL FUNCTIONS
   ========================================================= */

function closeTeamPanels() {

    const joinPanel = document.getElementById("joinTeamPanel");
    const createPanel = document.getElementById("createTeamPanel");

    if (joinPanel) {
        joinPanel.classList.remove("active");
        joinPanel.style.display = "none";
    }

    if (createPanel) {
        createPanel.classList.remove("active");
        createPanel.style.display = "none";
    }
}


function openJoinTeam() {

    const joinPanel = document.getElementById("joinTeamPanel");
    const createPanel = document.getElementById("createTeamPanel");

    if (!joinPanel) return;

    if (createPanel) {
        createPanel.style.display = "none";
        createPanel.classList.remove("active");
    }

    joinPanel.style.display = "block";

    setTimeout(() => {
        joinPanel.classList.add("active");
    }, 10);

    joinPanel.scrollIntoView({
        behavior: "smooth",
        block: "start"
    });
}


function openCreateTeam() {

    const joinPanel = document.getElementById("joinTeamPanel");
    const createPanel = document.getElementById("createTeamPanel");

    if (!createPanel) return;

    if (joinPanel) {
        joinPanel.style.display = "none";
        joinPanel.classList.remove("active");
    }

    createPanel.style.display = "block";

    setTimeout(() => {
        createPanel.classList.add("active");
    }, 10);

    createPanel.scrollIntoView({
        behavior: "smooth",
        block: "start"
    });
}


/* =========================================================
   JOIN TEAM
   ========================================================= */

function requestToJoinTeam(teamName) {

    const confirmed = confirm(
        "Send request to join " + teamName + "?"
    );

    if (!confirmed) {
        return;
    }

    // Find clicked button
    const buttons = document.querySelectorAll(".team-join-btn");

    buttons.forEach(button => {

        if (
            button.getAttribute("onclick") &&
            button.getAttribute("onclick").includes(teamName)
        ) {

            button.textContent = "Request Sent";
            button.disabled = true;
            button.classList.add("request-sent");

        }

    });

    alert(
        "Your request to join " +
        teamName +
        " has been sent successfully!"
    );
}


/* =========================================================
   CREATE TEAM
   ========================================================= */

function createTeam() {

    const nameInput =
        document.getElementById("newTeamName");

    const skillInput =
        document.getElementById("newTeamSkill");

    const descriptionInput =
        document.getElementById("newTeamDescription");

    if (!nameInput || !skillInput || !descriptionInput) {
        return;
    }

    const teamName = nameInput.value.trim();
    const skills = skillInput.value.trim();
    const description = descriptionInput.value.trim();

    // Validation
    if (teamName === "") {

        alert("Please enter a team name.");
        nameInput.focus();
        return;

    }

    if (skills === "") {

        alert("Please enter the skills required.");
        skillInput.focus();
        return;

    }

    if (description === "") {

        alert("Please describe what your team is building.");
        descriptionInput.focus();
        return;

    }


    // Create initials
    const initials = teamName
        .split(" ")
        .filter(word => word.length > 0)
        .slice(0, 2)
        .map(word => word[0].toUpperCase())
        .join("");


    // Create new team card
    const teamList =
        document.getElementById("teamList");

    if (teamList) {

        const newTeam = document.createElement("div");

        newTeam.className = "team-card";

        newTeam.setAttribute(
            "data-team-search",
            (
                teamName +
                " " +
                skills +
                " " +
                description
            ).toLowerCase()
        );

        newTeam.innerHTML = `

            <div class="team-card-main">

                <div class="team-avatar">
                    ${initials || "TM"}
                </div>

                <div>

                    <h3>${escapeHTML(teamName)}</h3>

                    <p>Created by You</p>

                    <div class="student-tags">

                        <span>${escapeHTML(skills)}</span>

                        <span>1/5 Members</span>

                    </div>

                </div>

            </div>

            <button
                class="connect-btn team-join-btn"
                disabled>
                Your Team
            </button>

        `;

        teamList.prepend(newTeam);
    }


    // Clear form
    nameInput.value = "";
    skillInput.value = "";
    descriptionInput.value = "";


    alert(
        "Team '" +
        teamName +
        "' has been created successfully!"
    );


    // Open join panel so user can see team
    openJoinTeam();
}


/* =========================================================
   TEAM SEARCH
   ========================================================= */

function filterTeams() {

    const searchInput =
        document.getElementById("teamSearch");

    const searchText =
        searchInput ?
        searchInput.value.toLowerCase().trim() :
        "";

    const teams =
        document.querySelectorAll(".team-card");

    teams.forEach(team => {

        const searchableText =
            team.getAttribute("data-team-search") ||
            team.textContent;

        if (
            searchableText
                .toLowerCase()
                .includes(searchText)
        ) {

            team.style.display = "";

        } else {

            team.style.display = "none";

        }

    });
}


/* =========================================================
   TEAM REQUESTS
   ========================================================= */

function handleTeamRequest(button, status) {

    if (!button) return;

    const requestCard =
        button.closest(".request-card");

    if (!requestCard) return;

    const studentNameElement =
        requestCard.querySelector(".request-info h3");

    const studentName =
        studentNameElement ?
        studentNameElement.textContent :
        "Student";


    if (status === "approved") {

        requestCard.innerHTML = `

            <div class="team-avatar">✓</div>

            <div class="request-info">

                <h3>${escapeHTML(studentName)}</h3>

                <p>
                    Request approved successfully.
                </p>

            </div>

            <div class="request-status approved">
                Approved
            </div>

        `;

    }


    if (status === "rejected") {

        requestCard.innerHTML = `

            <div class="team-avatar">×</div>

            <div class="request-info">

                <h3>${escapeHTML(studentName)}</h3>

                <p>
                    Request has been rejected.
                </p>

            </div>

            <div class="request-status rejected">
                Rejected
            </div>

        `;

    }

}


/* =========================================================
   DISCOVER STUDENTS - BACKEND / MYSQL
   ========================================================= */

let discoverStudents = [];


/* =========================================================
   GET AUTH TOKEN
   ========================================================= */

function getAuthToken() {
    return (
        localStorage.getItem("token") ||
        localStorage.getItem("campusConnectToken") ||
        localStorage.getItem("jwtToken") ||
        null
    );
}


/* =========================================================
   STUDENT INITIALS
   ========================================================= */

function getStudentInitials(name) {

    if (!name) return "CC";

    return name
        .split(" ")
        .filter(Boolean)
        .slice(0, 2)
        .map(word => word.charAt(0).toUpperCase())
        .join("");
}


/* =========================================================
   LOAD STUDENTS FROM BACKEND
   ========================================================= */

async function loadDiscoverStudents() {

    const studentGrid =
        document.querySelector(".student-grid");

    if (!studentGrid) return;

    const token = getAuthToken();

    if (!token) {

        studentGrid.innerHTML = `
            <div style="
                grid-column:1/-1;
                text-align:center;
                padding:40px;
                color:#64748b;
            ">
                Please login to discover students.
            </div>
        `;

        return;
    }


    /* LOADING */

    studentGrid.innerHTML = `
        <div style="
            grid-column:1/-1;
            text-align:center;
            padding:40px;
            color:#64748b;
        ">
            Loading students...
        </div>
    `;


    try {

        const response = await fetch(
            "https://campus-connect-backend-ruq3.onrender.com/api/users",
            {
                method: "GET",

                headers: {
                    "Authorization": "Bearer " + token,
                    "Content-Type": "application/json"
                }
            }
        );


        const data = await response.json();


        if (!response.ok) {
            throw new Error(
                data.message ||
                "Failed to fetch students"
            );
        }


        discoverStudents =
            Array.isArray(data.users)
                ? data.users
                : [];


        renderDiscoverStudents();


    } catch (error) {

        console.error(
            "Discover students error:",
            error
        );


        studentGrid.innerHTML = `
            <div style="
                grid-column:1/-1;
                text-align:center;
                padding:40px;
                color:#ef4444;
            ">
                Failed to load students.

                <br>

                <small>
                    Please make sure you are logged in
                    and backend is running.
                </small>
            </div>
        `;
    }
}

/* =========================================================
   STUDENT SEARCH + FILTER
   ========================================================= */

function filterStudents() {

    const searchInput =
        document.getElementById("studentSearch");

    const filterSelect =
        document.getElementById("studentFilter");


    const searchText =
        searchInput
            ? searchInput.value.toLowerCase().trim()
            : "";


    const selectedFilter =
        filterSelect
            ? filterSelect.value
            : "all";


    const students =
        document.querySelectorAll(".student-card");


    students.forEach(student => {

        const searchableText =
            student.getAttribute(
                "data-student-search"
            ) ||
            student.textContent;


        const text =
            searchableText.toLowerCase();


        const matchesSearch =
            text.includes(searchText);


        let matchesFilter = true;


        if (selectedFilter !== "all") {

            switch (selectedFilter) {

                case "web":

                    matchesFilter =
                        text.includes("web") ||
                        text.includes("react") ||
                        text.includes("node") ||
                        text.includes("javascript");

                    break;


                case "ai":

                    matchesFilter =
                        text.includes("ai") ||
                        text.includes("machine learning") ||
                        text.includes("ml") ||
                        text.includes("python");

                    break;


                case "design":

                    matchesFilter =
                        text.includes("design") ||
                        text.includes("ui") ||
                        text.includes("ux") ||
                        text.includes("figma") ||
                        text.includes("graphic");

                    break;


                case "dsa":

                    matchesFilter =
                        text.includes("dsa") ||
                        text.includes("programming") ||
                        text.includes("cpp") ||
                        text.includes("c++");

                    break;


                case "startup":

                    matchesFilter =
                        text.includes("startup") ||
                        text.includes("business") ||
                        text.includes("entrepreneur");

                    break;
            }
        }


        if (
            matchesSearch &&
            matchesFilter
        ) {

            student.style.display = "";

        } else {

            student.style.display = "none";

        }

    });
}

let communities = [];
let selectedCommunity = null;
let communityMembers = [];

const COMMUNITY_META = {
    "Coding & Programming": {category:"Technology",icon:"💻"},
    "AI & Machine Learning": {category:"Technology",icon:"🤖"},
    "Startup & Entrepreneurship": {category:"Career & Business",icon:"🚀"},
    "UI/UX & Design": {category:"Creative",icon:"🎨"},
    "Photography & Media": {category:"Creative",icon:"📸"},
    "Gaming Community": {category:"Entertainment",icon:"🎮"},
    "Fitness & Sports": {category:"Sports & Fitness",icon:"🏋️"}
};

function getCommunityMeta(name){
    return COMMUNITY_META[name] || {category:"Community",icon:"🌐"};
}

function getCurrentUserId(){
    try{
        const token=getAuthToken();
        if(!token)return null;
        const p=JSON.parse(atob(token.split(".")[1].replace(/-/g,"+").replace(/_/g,"/")));
        return p.userId || p.id || p.user_id || null;
    }catch(e){return null;}
}

function communityStore(type,id){
    return `campusConnect_${type}_${id}`;
}

function readCommunityStore(type,id,fallback=[]){
    try{
        const raw=localStorage.getItem(communityStore(type,id));
        return raw ? JSON.parse(raw) : fallback;
    }catch(e){return fallback;}
}

function writeCommunityStore(type,id,value){
    localStorage.setItem(communityStore(type,id),JSON.stringify(value));
}

/* ================= LOAD COMMUNITIES ================= */

async function loadCommunities(){
    const grid=document.getElementById("communityGrid");
    if(!grid)return;

    const token=getAuthToken();
    if(!token){
        grid.innerHTML=`<div class="empty-state"><div>🔐</div><h2>Please login first</h2><p>Login to view and join Campus Connect communities.</p></div>`;
        return;
    }

    grid.innerHTML=`<div class="empty-state"><div>⏳</div><h2>Loading communities...</h2></div>`;

    try{
        const response=await fetch("https://campus-connect-backend-ruq3.onrender.com/api/communities",{
            headers:{
                "Authorization":"Bearer "+token,
                "Content-Type":"application/json"
            }
        });
        const data=await response.json();
        if(!response.ok)throw new Error(data.message||"Failed to load communities");

        communities=(Array.isArray(data.communities)?data.communities:[]).map(c=>{
            const meta=getCommunityMeta(c.name);
            return {...c,category:meta.category,icon:meta.icon,members:Number(c.member_count||0),active:0};
        });

        renderCommunities();
    }catch(error){
        console.error("Load communities error:",error);
        grid.innerHTML=`<div class="empty-state"><div>⚠️</div><h2>Failed to load communities</h2><p>${escapeHTML(error.message||"Please try again.")}</p></div>`;
    }
}

/* ================= RENDER COMMUNITIES ================= */

function renderCommunities(){
    const grid=document.getElementById("communityGrid");
    if(!grid)return;

    const search=document.getElementById("communitySearch");
    const searchText=search ? search.value.toLowerCase().trim() : "";
    const active=document.querySelector(".category.active");
    const category=active ? active.getAttribute("data-category") : "All";

    const filtered=communities.filter(c=>{
        const text=`${c.name||""} ${c.description||""}`.toLowerCase();
        return text.includes(searchText) && (category==="All" || c.category===category);
    });

    grid.innerHTML="";
    if(!filtered.length){
        grid.innerHTML=`<div class="empty-state"><div>🔍</div><h2>No communities found</h2><p>Try another search or category.</p></div>`;
        return;
    }

    filtered.forEach(c=>{
        const card=document.createElement("div");
        card.className="community-card";
        card.innerHTML=`
            <div class="community-icon">${c.icon}</div>
            <span class="community-category">${escapeHTML(c.category)}</span>
            <h3>${escapeHTML(c.name)}</h3>
            <p>${escapeHTML(c.description||"No description provided.")}</p>
            <div class="community-card-stats">
                <span>👥 ${Number(c.members||0)} Members</span>
                <span>🟢 Community</span>
            </div>
            <button class="primary-btn" onclick="openCommunity(${Number(c.id)})">View Community</button>
        `;
        grid.appendChild(card);
    });
}

/* ================= OPEN COMMUNITY ================= */

async function openCommunity(communityId){
    selectedCommunity=communities.find(c=>Number(c.id)===Number(communityId));
    if(!selectedCommunity)return;

    const set=(id,value)=>{
        const el=document.getElementById(id);
        if(el)el.textContent=value;
    };

    set("detailName",selectedCommunity.name);
    set("detailCategory",selectedCommunity.category);
    set("detailMembers",selectedCommunity.members);
    set("detailActive","—");
    set("detailDescription",selectedCommunity.description||"No description provided.");
    set("detailIcon",selectedCommunity.icon);

    showPage("community-detail");
    openCommunityTab("about");

    await loadCommunityMembers(selectedCommunity.id);
    await loadCommunityDiscussions();
    await loadCommunityEvents();
}

/* ================= MEMBERS ================= */

async function loadCommunityMembers(communityId){
    const token=getAuthToken();
    const grid=document.getElementById("memberGrid");
    if(!token||!grid)return;

    grid.innerHTML=`<div style="grid-column:1/-1;text-align:center;padding:30px;color:#64748b;">Loading members...</div>`;

    try{
        const response=await fetch(`https://campus-connect-backend-ruq3.onrender.com/api/communities/${communityId}/members`,{
            headers:{
                "Authorization":"Bearer "+token,
                "Content-Type":"application/json"
            }
        });
        const data=await response.json();
        if(!response.ok)throw new Error(data.message||"Failed to load members");

        communityMembers=Array.isArray(data.members)?data.members:[];

        if(selectedCommunity)selectedCommunity.members=communityMembers.length;
        const count=document.getElementById("detailMembers");
        if(count)count.textContent=communityMembers.length;

        renderCommunityMembers();
        updateCommunityMembershipButton();
    }catch(error){
        console.error("Load members error:",error);
        grid.innerHTML=`<div style="grid-column:1/-1;text-align:center;padding:30px;color:#ef4444;">Failed to load members.<br><small>${escapeHTML(error.message||"")}</small></div>`;
    }
}

function renderCommunityMembers(){
    const grid=document.getElementById("memberGrid");
    if(!grid)return;

    grid.innerHTML="";
    if(!communityMembers.length){
        grid.innerHTML=`<div style="grid-column:1/-1;text-align:center;padding:30px;color:#64748b;">No members yet.</div>`;
        return;
    }

    const me=getCurrentUserId();

    communityMembers.forEach(member=>{
        const card=document.createElement("div");
        card.className="member-card";
        card.innerHTML=`
            <div class="student-avatar">${escapeHTML(getStudentInitials(member.name))}</div>
            <div style="flex:1;">
                <h3>${escapeHTML(member.name||"Student")} ${Number(member.id)===Number(me)?"<small>(You)</small>":""}</h3>
                <p>${escapeHTML(member.course||"Campus Student")}${member.year?" • Year "+escapeHTML(member.year):""}</p>
                ${member.college?`<small>${escapeHTML(member.college)}</small>`:""}
                ${member.email?`<small style="display:block;margin-top:4px;">${escapeHTML(member.email)}</small>`:""}
            </div>
        `;
        grid.appendChild(card);
    });
}

/* ================= JOIN / LEAVE ================= */

function updateCommunityMembershipButton(){
    const button=document.getElementById("joinCommunityBtn");
    if(!button||!selectedCommunity)return;

    const me=getCurrentUserId();
    const isMember=communityMembers.some(m=>Number(m.id)===Number(me));

    button.disabled=false;
    button.classList.remove("joined");
    button.textContent=isMember ? "Leave Community" : "Join Community";
    if(isMember)button.classList.add("joined");
}

async function joinSelectedCommunity(){
    const token=getAuthToken();
    const button=document.getElementById("joinCommunityBtn");
    if(!token||!selectedCommunity)return;

    button.disabled=true;
    button.textContent="Joining...";

    try{
        const response=await fetch(`https://campus-connect-backend-ruq3.onrender.com/api/communities/${selectedCommunity.id}/join`,{
            method:"POST",
            headers:{

   This keeps chat persistent per community in this browser

function loadGroupMessages(){
    const container=document.getElementById("groupMessages");
    if(!container||!selectedCommunity)return;

    const messages=readCommunityStore("groupMessages",selectedCommunity.id,[]);
    container.innerHTML="";

    if(!messages.length){
        container.innerHTML=`<div style="text-align:center;padding:25px;color:#64748b;">No messages yet. Start the conversation!</div>`;
        return;
    }

    messages.forEach(message=>{
        const div=document.createElement("div");
        div.className="group-message";
        div.innerHTML=`
            <strong>${escapeHTML(message.name)}</strong>
            <p>${escapeHTML(message.message)}</p>
            <small>${escapeHTML(message.time)}</small>
        `;
        container.appendChild(div);
    });

    container.scrollTop=container.scrollHeight;
}

function sendGroupMessage(){
    const input=document.getElementById("groupInput");
    if(!input||!selectedCommunity)return;

    const message=input.value.trim();
    if(!message)return;

    const messages=readCommunityStore("groupMessages",selectedCommunity.id,[]);
    messages.push({
        name:"You",
        message,
        time:new Date().toLocaleTimeString([],{
            hour:"2-digit",minute:"2-digit"
        })
    });

    writeCommunityStore("groupMessages",selectedCommunity.id,messages);
    input.value="";
    loadGroupMessages();
}

function getDiscussionActorId(){
    try{
        if(typeof getCurrentUserId === "function"){
            const id=getCurrentUserId();
            if(id!==null && id!==undefined && String(id).trim()!=="")return String(id);
        }
    }catch(error){
        console.warn("Could not read current user id:",error);
    }
    return "guest";
}

function normalizeDiscussionPost(post){
    if(!post||typeof post!=="object")return post;

    if(!Array.isArray(post.likes))post.likes=[];
    if(!Array.isArray(post.comments))post.comments=[];

    return post;
}

function toggleDiscussionLike(postIndex){
    if(!selectedCommunity)return;

    const posts=readCommunityStore("discussions",selectedCommunity.id,[]).map(normalizeDiscussionPost);
    const post=posts[postIndex];
    if(!post)return;

    const actor=getDiscussionActorId();
    const liked=post.likes.some(id=>String(id)===actor);

    if(liked){
        post.likes=post.likes.filter(id=>String(id)!==actor);
    }else{
        post.likes.push(actor);
    }

    writeCommunityStore("discussions",selectedCommunity.id,posts);
    renderCommunityDiscussions(posts);
}

function toggleDiscussionComments(postIndex){
    const box=document.getElementById(`comments-${postIndex}`);
    if(!box)return;

    const isHidden=box.style.display==="none";
    box.style.display=isHidden?"block":"none";

    if(isHidden){
        const input=box.querySelector(".discussion-comment-input");
        if(input)setTimeout(()=>input.focus(),50);
    }
}

function addDiscussionComment(postIndex){
    if(!selectedCommunity)return;

    const input=document.getElementById(`comment-input-${postIndex}`);
    if(!input)return;

    const text=input.value.trim();
    if(!text){
        input.focus();
        return;
    }

    const posts=readCommunityStore("discussions",selectedCommunity.id,[]).map(normalizeDiscussionPost);
    const post=posts[postIndex];
    if(!post)return;

    post.comments.push({
        id:`comment-${Date.now()}-${Math.random().toString(36).slice(2,8)}`,
        author:"You",
        text,
        time:new Date().toLocaleString()
    });

    writeCommunityStore("discussions",selectedCommunity.id,posts);
    renderCommunityDiscussions(posts);

    setTimeout(()=>{
        const box=document.getElementById(`comments-${postIndex}`);
        const newInput=document.getElementById(`comment-input-${postIndex}`);
        if(box)box.style.display="block";
        if(newInput)newInput.focus();
    },50);
}

function renderCommunityDiscussions(posts){
    const container=document.getElementById("fullDiscussions");
    if(!container||!selectedCommunity)return;

    const data=(posts||readCommunityStore("discussions",selectedCommunity.id,[])).map(normalizeDiscussionPost);
    container.innerHTML="";

    if(!data.length){
        container.innerHTML=`<div style="padding:25px;text-align:center;color:#64748b;">No discussions yet. Start one!</div>`;
        renderCommunityOverview([]);
        return;
    }

    const actor=getDiscussionActorId();

    data.forEach((post,index)=>{
        const liked=post.likes.some(id=>String(id)===actor);
        const likeCount=post.likes.length;
        const commentCount=post.comments.length;

        const div=document.createElement("div");
        div.className="discussion-preview discussion-post";
        div.innerHTML=`
            <div style="display:flex;justify-content:space-between;align-items:flex-start;gap:12px;">
                <strong class="post-author">${escapeHTML(post.author||"You")}</strong>
                ${post.time?`<small class="post-time">${escapeHTML(post.time)}</small>`:""}
            </div>
            <p>${escapeHTML(post.text)}</p>

            <div class="post-actions" style="display:flex;gap:10px;align-items:center;margin-top:12px;">
                <button type="button"
                    onclick="toggleDiscussionLike(${index})"
                    style="border:1px solid #dbe2ea;background:${liked?"#eef3ff":"#fff"};color:${liked?"#173f8f":"#475569"};padding:7px 12px;border-radius:8px;cursor:pointer;font-weight:600;">
                    ${liked?"❤️ Liked":"♡ Like"} <span>${likeCount}</span>
                </button>
                <button type="button"
                    onclick="toggleDiscussionComments(${index})"
                    style="border:1px solid #dbe2ea;background:#fff;color:#475569;padding:7px 12px;border-radius:8px;cursor:pointer;font-weight:600;">
                    💬 Comments <span>${commentCount}</span>
                </button>
            </div>

            <div id="comments-${index}" style="display:none;margin-top:14px;padding-top:14px;border-top:1px solid #e5e7eb;">
                <div style="display:flex;gap:8px;margin-bottom:12px;">
                    <input
                        id="comment-input-${index}"
                        class="discussion-comment-input"
                        type="text"
                        placeholder="Write a comment..."
                        onkeydown="if(event.key==='Enter'){event.preventDefault();addDiscussionComment(${index});}"
                        style="flex:1;border:1px solid #dbe2ea;border-radius:8px;padding:10px 12px;outline:none;">
                    <button type="button"
                        onclick="addDiscussionComment(${index})"
                        style="border:none;background:#173f8f;color:#fff;padding:0 15px;border-radius:8px;font-weight:600;cursor:pointer;">
                        Comment
                    </button>
                </div>

                <div class="discussion-comments-list">
                    ${commentCount?post.comments.map(comment=>`
                        <div style="padding:10px 12px;background:#f8fafc;border-radius:8px;margin-bottom:8px;">
                            <div style="display:flex;justify-content:space-between;gap:10px;">
                                <strong style="font-size:13px;">${escapeHTML(comment.author||"You")}</strong>
                                <small style="color:#94a3b8;">${escapeHTML(comment.time||"")}</small>
                            </div>
                            <div style="margin-top:5px;color:#334155;">${escapeHTML(comment.text||"")}</div>
                        </div>
                    `).join(""):"<div style=\"color:#94a3b8;font-size:13px;\">No comments yet. Start the discussion.</div>"}
                </div>
            </div>
        `;
        container.appendChild(div);
    });

    renderCommunityOverview(data);
}

function createDiscussionPost(){
    const input=document.getElementById("newPostInput");
    if(!input||!selectedCommunity)return;

    const text=input.value.trim();
    if(!text){
        alert("Please write something first.");
        input.focus();
        return;
    }

    const posts=readCommunityStore("discussions",selectedCommunity.id,[]).map(normalizeDiscussionPost);
    posts.unshift({
        author:"You",
        text,
        time:new Date().toLocaleString(),
        likes:[],
        comments:[]
    });

    writeCommunityStore("discussions",selectedCommunity.id,posts);
    input.value="";
    renderCommunityDiscussions(posts);
    alert("Discussion posted successfully!");
}

function escapeHTML(value) {

    return String(value)

        .replace(/&/g, "&amp;")

        .replace(/</g, "&lt;")

        .replace(/>/g, "&gt;")

        .replace(/"/g, "&quot;")

        .replace(/'/g, "&#039;");

}

function renderChatMessages(messages) {

    const chatMessages =
        document.getElementById("chatMessages");

    if (!chatMessages) return;

    if (!messages.length) {

        chatMessages.innerHTML = `
            <div class="chat-placeholder">
                <div>💬</div>
                <h3>No messages yet</h3>
                <p>Send the first message.</p>
            </div>
        `;

        return;
    }

    const currentUserId = getCurrentUserId();

    console.log("Current User ID:", currentUserId);

    chatMessages.innerHTML = messages.map(message => {

        const isSent =
            Number(message.sender_id) === Number(currentUserId);

        const time = message.created_at
            ? new Date(message.created_at).toLocaleTimeString(
                [],
                {
                    hour: "2-digit",
                    minute: "2-digit"
                }
            )
            : "";

        return `
            <div class="message-bubble ${isSent ? "sent" : "received"}">

                ${escapeMessageHTML(message.message)}

                <span class="message-time">
                    ${time}
                </span>

            </div>
        `;

    }).join("");

    chatMessages.scrollTop =
        chatMessages.scrollHeight;
}

        const response = await fetch(

            {
                method: "POST",

                headers: {
                    "Authorization": "Bearer " + token,
                    "Content-Type": "application/json"
                },

                body: JSON.stringify({
                    receiverId: activeChatUserId,
                    message: message
                })
            }
        );

        const data = await response.json();

        if (!response.ok) {
            throw new Error(
                data.message || "Message could not be sent"
            );
        }

        input.value = "";

        await loadConversation(
            activeChatUserId,
            activeChatUserName
        );

    } catch (error) {

        console.error("Send message error:", error);

        alert(error.message);

    } finally {

        sendButton.disabled = false;

        input.focus();
    }
}


/* ================= SAFE MESSAGE TEXT ================= */

function escapeMessageHTML(text) {

    const div = document.createElement("div");

    div.textContent = text;

    return div.innerHTML;
}


/* ================= CURRENT USER ID ================= */

function getCurrentUserId() {

    const token = getAuthToken();

    if (!token) {
        return null;
    }

    try {

        const payload =
            JSON.parse(
                atob(token.split(".")[1])
            );

        return Number(
            payload.userId ||
            payload.id ||
            payload.sub
        );

    } catch (error) {

        console.error(
            "Could not decode JWT:",
            error
        );

        return null;
    }
}

        chatUserStatus.textContent =
            "Private conversation";
    }

    if (chatMessages) {
        chatMessages.innerHTML = `
            <div class="chat-placeholder">
                <div>⏳</div>
                <p>Loading messages...</p>
            </div>
        `;
    }

    try {

        const response = await fetch(
            "https://campus-connect-backend-ruq3.onrender.com/api/messages/" +
            activeChatUserId,
            {
                method: "GET",
                headers: {
                    "Authorization":
                        "Bearer " + token,
                    "Content-Type":
                        "application/json"
                }
            }
        );

        const data = await response.json();

        if (!response.ok) {
            throw new Error(
                data.message ||
                "Failed to load conversation"
            );
        }

        renderChatMessages(
            data.messages || []
        );

    } catch (error) {

        console.error(
            "Conversation loading error:",
            error
        );

        if (chatMessages) {
            chatMessages.innerHTML = `
                <div class="chat-placeholder">
                    <div>⚠️</div>
                    <p>
                        ${escapeHTML(error.message)}
                    </p>
                </div>
            `;
        }

    }

}


/* RENDER MESSAGES */

function renderChatMessages(messages) {

    const chatMessages =
        document.getElementById("chatMessages");

    if (!chatMessages) return;

    if (!messages.length) {

        chatMessages.innerHTML = `
            <div class="chat-placeholder">
                <div>💬</div>

                <h3>
                    Start a conversation
                </h3>

                <p>
                    Send the first message.
                </p>
            </div>
        `;

        return;
    }

    const currentUserId =
        getCurrentUserId();

    console.log(
        "Current User ID:",
        currentUserId
    );

    chatMessages.innerHTML =
        messages.map(message => {

            const isSent =
                Number(message.sender_id) ===
                Number(currentUserId);

            const time =
                message.created_at
                    ? new Date(
                        message.created_at
                    ).toLocaleTimeString(
                        [],
                        {
                            hour: "2-digit",
                            minute: "2-digit"
                        }
                    )
                    : "";

            return `
                <div class="message-bubble ${
                    isSent
                        ? "sent"
                        : "received"
                }">

                    ${escapeMessageHTML(
                        message.message
                    )}

                    <span class="message-time">
                        ${time}
                    </span>

                </div>
            `;

        }).join("");

    chatMessages.scrollTop =
        chatMessages.scrollHeight;
}


/* SEND MESSAGE */

async function sendMessage() {

    const input =
        document.getElementById(
            "messageInput"
        );

    if (!input) return;

    const message =
        input.value.trim();

    if (!message) return;

    if (!activeChatUserId) {

        alert(
            "Please select a conversation first."
        );

        return;
    }

    const token =
        getAuthToken();

    if (!token) {

        alert("Please login first.");

        return;
    }

    const sendButton =
        document.getElementById(
            "sendMessageBtn"
        );

    if (sendButton) {
        sendButton.disabled = true;
    }

    try {

        const response =
            await fetch(
                "https://campus-connect-backend-ruq3.onrender.com/api/messages",
                {
                    method: "POST",

                    headers: {
                        "Authorization":
                            "Bearer " + token,

                        "Content-Type":
                            "application/json"
                    },

                    body: JSON.stringify({
                        receiverId:
                            Number(
                                activeChatUserId
                            ),

                        message:
                            message
                    })
                }
            );

        const data =
            await response.json();

        if (!response.ok) {

            throw new Error(
                data.message ||
                "Failed to send message"
            );

        }

        input.value = "";

        await loadConversation(
            activeChatUserId,
            activeChatUserName
        );

    } catch (error) {

        console.error(
            "Send message error:",
            error
        );

        alert(
            error.message ||
            "Failed to send message."
        );

    } finally {

        if (sendButton) {
            sendButton.disabled = false;
        }

    }

}

/* ESCAPE MESSAGE HTML */

function escapeMessageHTML(value) {

    return String(value)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}
/* =========================================================
   ADMIN PANEL
   ========================================================= */

function getJwtPayload() {
    try {
        const token = getAuthToken();

        if (!token) {
            return null;
        }

        const parts = token.split(".");

        if (parts.length !== 3) {
            return null;
        }

        return JSON.parse(atob(parts[1]));
    } catch (error) {
        console.error("JWT decode error:", error);
        return null;
    }
}

        adminNavBtn.style.display = "inline-flex";

    }
}

   ADMIN STATS
   ========================= */

async function loadAdminStats() {
    const token = getAuthToken();

    if (!token) {
        console.error("No authentication token found.");
        return;
    }

    try {
        const response = await fetch("https://campus-connect-backend-ruq3.onrender.com/api/admin/stats", {
            method: "GET",
            headers: {
                "Authorization": `Bearer ${token}`,
                "Content-Type": "application/json"
            }
        });

        const data = await response.json();

        if (!response.ok) {
            console.error("Admin stats error:", data);
            return;
        }

        if (!data.success || !data.stats) {
            console.error("Invalid admin stats response:", data);
            return;
        }

        const stats = data.stats;

        const totalUsers = document.getElementById("adminTotalUsers");
        const totalCommunities = document.getElementById("adminTotalCommunities");
        const totalConnections = document.getElementById("adminTotalConnections");
        const totalMessages = document.getElementById("adminTotalMessages");

        if (totalUsers) {
            totalUsers.textContent = stats.totalUsers ?? 0;
        }

        if (totalCommunities) {
            totalCommunities.textContent = stats.totalCommunities ?? 0;
        }

        if (totalConnections) {
            totalConnections.textContent = stats.totalConnections ?? 0;
        }

        if (totalMessages) {
            totalMessages.textContent = stats.totalMessages ?? 0;
        }

        console.log("Admin stats loaded:", stats);

    } catch (error) {
        console.error("Failed to load admin stats:", error);
    }
}

/* =========================
   LOAD ADMIN USERS
   ========================= */

async function loadAdminUsers() {
    const token = getAuthToken();

    if (!token) {
        console.error("No authentication token found.");
        return;
    }

    const searchInput = document.getElementById("adminUserSearch");
    const tableContainer = document.getElementById("adminUsersTableContainer");
    const countElement = document.getElementById("adminUsersCount");

    const search = searchInput
        ? searchInput.value.trim()
        : "";

    if (tableContainer) {
        tableContainer.innerHTML = `
            <div class="admin-users-loading">
                Loading users...
            </div>
        `;
    }

    try {
        let url = "https://campus-connect-backend-ruq3.onrender.com/api/admin/users";

        if (search) {
            url += `?search=${encodeURIComponent(search)}`;
        }

        const response = await fetch(url, {
            method: "GET",
            headers: {
                "Authorization": `Bearer ${token}`,
                "Content-Type": "application/json"
            }
        });

        const data = await response.json();

        if (!response.ok) {
            throw new Error(data.message || "Failed to load users.");
        }

        if (!data.success) {
            throw new Error(data.message || "Failed to load users.");
        }

        renderAdminUsers(data.users || []);

    } catch (error) {
        console.error("Admin users error:", error);

        if (tableContainer) {
            tableContainer.innerHTML = `
                <div class="admin-users-error">
                    Failed to load users.
                </div>
            `;
        }

        if (countElement) {
            countElement.textContent = "Unable to load users";
        }
    }
}


/* =========================
   RENDER ADMIN USERS
   ========================= */

function renderAdminUsers(users) {
    const tableContainer = document.getElementById(
        "adminUsersTableContainer"
    );

    const countElement = document.getElementById(
        "adminUsersCount"
    );

    if (!tableContainer) {
        return;
    }

    if (countElement) {
        countElement.textContent =
            `${users.length} user${users.length === 1 ? "" : "s"} found`;
    }

    if (!users.length) {
        tableContainer.innerHTML = `
            <div class="admin-users-empty">
                No users found.
            </div>
        `;

        return;
    }

    let rows = "";

    users.forEach((user) => {

        const roleClass =
            user.role === "admin"
                ? "admin-role"
                : "student-role";

        rows += `
            <tr>
                <td>
                    <strong>${escapeAdminHTML(user.name || "N/A")}</strong>
                </td>

                <td>
                    ${escapeAdminHTML(user.email || "N/A")}
                </td>

                <td>
                    ${escapeAdminHTML(user.college || "N/A")}
                </td>

                <td>
                    ${escapeAdminHTML(user.course || "N/A")}
                </td>

                <td>
                    ${escapeAdminHTML(user.year || "N/A")}
                </td>

                <td>
                    <span class="admin-user-role ${roleClass}">
                        ${escapeAdminHTML(user.role || "student")}
                    </span>
                </td>
            </tr>
        `;
    });

    tableContainer.innerHTML = `
        <div class="admin-table-wrapper">
            <table class="admin-users-table">

                <thead>
                    <tr>
                        <th>Name</th>
                        <th>Email</th>
                        <th>College</th>
                        <th>Course</th>
                        <th>Year</th>
                        <th>Role</th>
                    </tr>
                </thead>

                <tbody>
                    ${rows}
                </tbody>

            </table>
        </div>
    `;
}
